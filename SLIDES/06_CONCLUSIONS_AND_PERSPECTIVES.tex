\begin{frame}{General Conclusions}

\vspace{0.15cm}

{\color{champsBlue}\bfseries
Lagrangian Droplet Impingement}

\vspace{0.10cm}

\begin{itemize}
    \item Efficient and scalable particle tracking,
    with adaptive seeding and accelerated intersections.
    \item Consistent Eulerian--Lagrangian predictions,
    with extended capabilities for SLD modeling.
\end{itemize}

\vspace{0.22cm}

{\color{champsBlue}\bfseries
Three-Dimensional Level-Set Ice Accretion}

\vspace{0.10cm}

\begin{itemize}
    \item Robust multilayer geometry evolution,
    avoiding surface-mesh entanglement.
    \item Successful prediction of complex 3D ice
    morphologies, including scallop structures.
\end{itemize}

\vspace{0.22cm}

{\color{champsBlue}\bfseries
Immersed Boundary Method}

\vspace{0.10cm}

\begin{itemize}
    \item Integration of flow, droplet, thermodynamic,
    and geometry-evolution models.
    \item Demonstrated feasibility of ice accretion
    without repeated body-fitted aerodynamic remeshing.
\end{itemize}

\begin{frame}{Current Limitations}

\vspace{0.15cm}

{\color{champsBlue}\bfseries Physical Modeling}

\vspace{0.10cm}
\begin{itemize}
    \item Improved roughness, ice-density, and near-wall
    heat-transfer models are needed for more accurate
    ice-shape predictions, especially for glaze cases.
\end{itemize}

\vspace{0.30cm}

{\color{champsBlue}\bfseries Lagrangian Resource Requirements}

\vspace{0.10cm}
\begin{itemize}
    \item Replication of aerodynamic mesh and flow data
    increases per-node memory requirements and
    limits applicability to very large meshes.
\end{itemize}

\vspace{0.30cm}

{\color{champsBlue}\bfseries Immersed Boundary Method}

\vspace{0.10cm}
\begin{itemize}
    \item The absence of adaptive mesh refinement
    requires large predefined refined regions,
    increasing computational cost, and near-wall
    heat-transfer predictions require further improvement.
\end{itemize}

\end{frame}

\begin{frame}{Future Perspectives}

\vspace{0.15cm}

{\color{champsBlue}\bfseries Physical Modeling}

\vspace{0.10cm}
\begin{itemize}
    \item Develop variable roughness and ice-density models, with improved heat-transfer predictions.
\end{itemize}

\vspace{0.25cm}

{\color{champsBlue}\bfseries Lagrangian Droplet Tracking}

\vspace{0.10cm}
\begin{itemize}
    \item Explore GPU acceleration with improved data locality and workload balancing.
\end{itemize}

\vspace{0.25cm}

{\color{champsBlue}\bfseries Level-Set Geometry Evolution}

\vspace{0.10cm}
\begin{itemize}
    \item Accelerate surface extraction, improve mesh quality, and investigate local mass conservation.
\end{itemize}

\vspace{0.25cm}

{\color{champsBlue}\bfseries Immersed Boundary Method}

\vspace{0.10cm}
\begin{itemize}
    \item Incorporate adaptive mesh refinement and improve near-wall heat-transfer treatment.
\end{itemize}

\end{frame}
